\begin{frame}{Proposition 1: dynamic equilibrium}
\small
\begin{block}{Exact selected statement}
Fix $X_1\ge0$ implied by the prior. There exists a unique symmetric Perfect Bayesian equilibrium path with $e_{i,t}=e_t\ge0$, where
\[
e_t=e(X_t,\tau_A),\qquad
X_{t+1}=\left[\Sigma^2+(X_t+\lambda_GIe_t)^{-1}\right]^{-1}.
\]
\end{block}
\textbf{Conditions doing the work:} atomistic agents, equal-sized islands, Gaussian signals, unique static best response, common parameters, and a fixed AI precision.
\paperref{Proposition 1, equation (7).}
\end{frame}

\begin{frame}[fragile]{Lean: what is actually verified}
\small
\begin{columns}[T,onlytextwidth]
\column{.47\textwidth}
\textbf{Mathematical claim}
\[
\Delta_I>0,\ \lambda_I>0,\ g(Y)>0
\Rightarrow \lambda_Ig(Y)\Delta_I>0.
\]
At $e=X=0$, the marginal benefit is positive; a zero-effort optimizer is impossible under the paper's monotone marginal structure.
\column{.49\textwidth}
\textbf{Lean boundary}
\begin{lstlisting}
theorem positive_standalone_return_at_zero
    (hDI : 0 < deltaI)
    (hLI : 0 < lambdaI)
    (hg  : 0 < gY) :
    0 < lambdaI * gY * deltaI := by
  positivity
\end{lstlisting}
\begin{block}{Interpretation}Lean verifies the ordered-ring implication. EconCSLib separately audits whether this statement faithfully represents the paper and extension.\end{block}
\end{columns}
\paperref{Repository \texttt{lean/OriginalExtension.lean}; original extension, not a paper-attributed theorem.}
\end{frame}
